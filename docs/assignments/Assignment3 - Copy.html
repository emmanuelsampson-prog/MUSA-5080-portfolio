  poverty_2010 <- poverty_2010 %>%
  mutate(
    poverty_rate = (povertyE / total_popE * 100),
    poverty_rate_moe = moe_prop(povertyE, total_popE, povertyM, total_popM)*100
  )
  poverty_2017 <- poverty_2017 %>%
  mutate(
    poverty_rate = (povertyE / total_popE * 100),
    poverty_rate_moe = moe_prop(povertyE, total_popE, povertyM, total_popM)*100
  )
poverty_2010 <- poverty_2010 %>%
  mutate(moe_pct = poverty_rate_moe / poverty_rate * 100)

poverty_2010 <- poverty_2010 %>%
  mutate(reliability = case_when(
    moe_pct < 12  ~ "Reliable",
    moe_pct < 40 ~ "Somewhat reliable",
    TRUE         ~ "Unreliable"
  ))
poverty_2017 <- poverty_2017 %>%
  mutate(moe_pct = poverty_rate_moe / poverty_rate * 100)

poverty_2017 <- poverty_2017 %>%
  mutate(reliability = case_when(
    moe_pct < 12  ~ "Reliable",
    moe_pct < 40 ~ "Somewhat reliable",
    TRUE         ~ "Unreliable"
  ))
poverty_2010 <- poverty_2010 %>%
  mutate(period = "2006-2010")

poverty_2017 <- poverty_2017 %>%
  mutate(period = "2013-2017")
poverty_change <- bind_rows(
  poverty_2010,
  poverty_2017
)
  poverty_wide <- poverty_change %>%
  select(
    GEOID,
    NAME,
    period,
    poverty_rate,
    poverty_rate_moe
  ) %>%
  pivot_wider(
    names_from = period,
    values_from = c(
      poverty_rate,
      poverty_rate_moe
    )
  )
